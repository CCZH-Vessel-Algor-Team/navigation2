cmake_minimum_required(VERSION 3.8)
project(usv_autonomy_bt)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
find_package(usv_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
add_library(usv_navigation_contract_bt_node SHARED src/navigation_contract.cpp)
target_compile_definitions(usv_navigation_contract_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(usv_navigation_contract_bt_node rclcpp behaviortree_cpp_v3 usv_interfaces)
add_executable(task_tree src/task_tree.cpp)
ament_target_dependencies(task_tree rclcpp behaviortree_cpp_v3 usv_interfaces std_msgs ament_index_cpp)
install(TARGETS usv_navigation_contract_bt_node LIBRARY DESTINATION lib)
install(TARGETS task_tree DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY trees DESTINATION share/${PROJECT_NAME})
ament_package()
